package PACKAGE

import (
	"encoding/json"
	"os"
	"testing"
)

func TestFrozenFidelityHistory(t *testing.T) {
	data, err := os.ReadFile(os.Getenv("FIDELITY_HISTORY"))
	if err != nil {
		t.Fatal(err)
	}
	result := fidelityHistory(t, data)
	output, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("FIDELITY_OUTPUT")+".reads", output, 0600); err != nil {
		t.Fatal(err)
	}
}
