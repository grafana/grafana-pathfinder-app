func fidelityWrite(t *testing.T, store *fakeAttemptStore, step fidelityStep, body map[string]any) *httptest.ResponseRecorder {
	store.createErr = nil
	if failure := fidelityFailure(step.Failure); failure != nil {
		store.createErr = k8s.NewServerResponseError(failure, 503)
	}
	if step.Status != 0 {
		status := metav1.Status{Code: int32(step.Status), Message: "injected"}
		if len(step.ErrorBody) > 0 {
			if err := json.Unmarshal(step.ErrorBody, &status); err != nil {
				t.Fatal(err)
			}
		}
		store.createErr = &apierrors.StatusError{ErrStatus: status}
	}
	store.replaceErrs = nil
	for i := 0; i < step.Conflicts; i++ {
		store.replaceErrs = append(store.replaceErrs, &apierrors.StatusError{ErrStatus: metav1.Status{Code: 409}})
	}
	raw, _ := json.Marshal(body)
	ctx := authlib.WithAuthInfo(context.Background(), completionIdentity{testIdentity: testIdentity{namespace: "stacks-1"}, subject: "user:abc"})
	handler := completionWriteHandler(store, func(context.Context, string, services.CompletionRecord) {})
	w := httptest.NewRecorder()
	if err := handler(ctx, w, &app.CustomRouteRequest{Method: http.MethodPost, Body: io.NopCloser(bytes.NewReader(raw)), ResourceIdentifier: resource.FullIdentifier{Namespace: "stacks-1"}}); err != nil {
		t.Fatal(err)
	}
	return w
}

func fidelityHistory(t *testing.T, data []byte) any {
	var corpus struct {
		Records     []services.CompletionRecord `json:"records"`
		Assignments []services.Assignment       `json:"assignments"`
	}
	if err := json.Unmarshal(data, &corpus); err != nil {
		t.Fatal(err)
	}
	complete, partial := services.CollateCompletions(corpus.Records)
	from := time.Date(2026, 9, 1, 0, 0, 0, 0, time.UTC)
	to := from.AddDate(0, 1, 0)
	evaluator := &services.ObligationEvaluator{Logger: log.DefaultLogger}
	reports := []any{}
	for _, historyComplete := range []bool{true, false} {
		report := services.NewLearningInsightsResponse(from, to)
		services.ShapeLearningInsights(&report, corpus.Records, corpus.Assignments, from, to, to, historyComplete, evaluator.GuideProgress)
		reports = append(reports, report)
	}
	return map[string]any{"completed": complete, "partial": partial, "reports": reports}
}

func newFidelityCache(t *testing.T) fidelityCacheDriver {
	services.ResetCompletionCache()
	t.Cleanup(services.ResetCompletionCache)
	now := time.Date(2026, 11, 1, 0, 0, 0, 0, time.UTC)
	key := services.CallerCompletionCacheKey("stacks-1", "user:abc")
	calls := 0
	return fidelityCacheDriver{
		advance:    func(duration time.Duration) { now = now.Add(duration) },
		invalidate: func() { services.InvalidateCompletionIndex("stacks-1") },
		read: func(forced bool, status int) fidelityCacheResult {
			index, err := services.LoadCompletionIndex(context.Background(), key, forced, log.DefaultLogger,
				func(context.Context) (*services.CompletionIndex, int, error) {
					calls++
					if status != 0 {
						return nil, 0, &apierrors.StatusError{ErrStatus: metav1.Status{Code: int32(status)}}
					}
					return &services.CompletionIndex{AsOf: now}, 1, nil
				}, func() time.Time { return now },
				func(err error) bool { code := sdkErrorStatus(err); return code >= 400 && code != 401 && code != 403 },
				func(error) string { return "upstream-error" })
			result := fidelityCacheResult{Calls: calls, Failed: err != nil}
			if index != nil {
				result.AsOf = index.AsOf.Format(time.RFC3339Nano)
			}
			return result
		},
	}
}

type fidelityReadTransport struct{ status int }

func (transport fidelityReadTransport) RoundTrip(request *http.Request) (*http.Response, error) {
	if transport.status == -4 {
		return nil, fidelityFailure("cancelled")
	}
	if transport.status == -5 {
		return nil, fidelityFailure("network")
	}
	if transport.status == -2 {
		return nil, context.DeadlineExceeded
	}
	status := transport.status
	body := []byte(`{"metadata":{},"items":[]}`)
	if status == -3 {
		status = 200
		records, token := fidelityReadPage(request.URL.Query().Get("continue"))
		items := []any{}
		for _, record := range records {
			items = append(items, map[string]any{"metadata": map[string]any{}, "spec": record})
		}
		body, _ = json.Marshal(map[string]any{"metadata": map[string]any{"continue": token}, "items": items})
	} else if status == -1 {
		status = 200
		body = []byte("not-json")
	} else if status != 200 {
		body, _ = json.Marshal(map[string]any{"kind": "Status", "apiVersion": "v1", "status": "Failure", "code": status, "message": "injected"})
	}
	return &http.Response{StatusCode: status, Header: http.Header{"Content-Type": []string{"application/json"}}, Body: io.NopCloser(bytes.NewReader(body)), Request: request}, nil
}

func fidelitySettingsRead(t *testing.T, status int, body string) *httptest.ResponseRecorder {
	upstream := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(status)
		_, _ = w.Write([]byte(body))
	}))
	defer upstream.Close()
	store, err := resourceaccess.NewSettingsStore(sdkplugin.BuildClientGenerator(resourceaccess.SettingsConfig(rest.Config{Host: upstream.URL, APIPath: "/apis"})))
	if err != nil {
		t.Fatal(err)
	}
	ctx := authlib.WithAuthInfo(context.Background(), testIdentity{namespace: "stacks-1"})
	response := httptest.NewRecorder()
	if err := settingsReadHandler(store.Read)(ctx, response, &app.CustomRouteRequest{Method: http.MethodGet, ResourceIdentifier: resource.FullIdentifier{Namespace: "stacks-1"}}); err != nil {
		t.Fatal(err)
	}
	return response
}
