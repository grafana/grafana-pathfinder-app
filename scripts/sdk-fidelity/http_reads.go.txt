package PACKAGE

import (
	"encoding/json"
	"fmt"
	"os"
	"testing"
	"time"
)

func TestFrozenFidelityHTTPReads(t *testing.T) {
	results := map[string]any{}
	for _, status := range []int{200, 400, 401, 403, 404, 405, 408, 409, 422, 429, 500, 501, 503, 504, -1, -2} {
		for _, capability := range []bool{false, true} {
			t.Run(fmt.Sprintf("%d/capability=%t", status, capability), func(t *testing.T) {
				response := fidelityCompletionRead(t, status, capability)
				var body map[string]any
				if err := json.Unmarshal(response.Body.Bytes(), &body); err != nil {
					t.Fatal(err)
				}
				if value, ok := body["asOf"]; ok {
					stamp, ok := value.(string)
					if !ok {
						t.Fatal("non-string asOf")
					}
					if _, err := time.Parse(time.RFC3339, stamp); err != nil {
						t.Fatal(err)
					}
					body["asOf"] = "<validated-server-time>"
				}
				results[fmt.Sprintf("%d/capability=%t", status, capability)] = map[string]any{
					"status": response.Code, "body": body,
					"contentType":  response.Header().Get("Content-Type"),
					"retryAfter":   response.Header().Get("Retry-After"),
					"cacheControl": response.Header().Get("Cache-Control"),
				}
			})
		}
	}
	data, err := json.MarshalIndent(results, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("FIDELITY_OUTPUT")+".httpreads", data, 0600); err != nil {
		t.Fatal(err)
	}
}
