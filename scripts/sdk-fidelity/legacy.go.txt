func fidelityWrite(t *testing.T, store *fakeAttemptStore, step fidelityStep, body map[string]any) *httptest.ResponseRecorder {
	store.createErr = fidelityFailure(step.Failure)
	if step.Status != 0 {
		store.createErr = &appPlatformUpstreamError{status: step.Status, msg: "injected", body: step.ErrorBody}
	}
	store.replaceErrs = nil
	for i := 0; i < step.Conflicts; i++ {
		store.replaceErrs = append(store.replaceErrs, &appPlatformUpstreamError{status: 409, msg: "injected"})
	}
	withCreator(t, store)
	r := writeRequest(t, "user:abc", body, testGrafanaConfig())
	// Model a valid stack/org pair; the baseline fixture's org 7 is deliberately synthetic.
	r = r.WithContext(backend.WithPluginContext(r.Context(), backend.PluginContext{Namespace: "stacks-1", OrgID: 1}))
	return doWrite(t, nil, r)
}

func fidelityHistory(t *testing.T, data []byte) any {
	var corpus struct {
		Records     []completionRecordSpec `json:"records"`
		Assignments []assignmentSpec       `json:"assignments"`
	}
	if err := json.Unmarshal(data, &corpus); err != nil {
		t.Fatal(err)
	}
	complete, partial := collateCompletions(corpus.Records)
	from := time.Date(2026, 9, 1, 0, 0, 0, 0, time.UTC)
	to := from.AddDate(0, 1, 0)
	evaluator := &obligationEvaluator{logger: log.DefaultLogger}
	reports := []any{}
	for _, historyComplete := range []bool{true, false} {
		report := newLearningInsightsResponse(from, to)
		shapeLearningInsights(&report, corpus.Records, corpus.Assignments, from, to, to, historyComplete, evaluator.guideProgress)
		reports = append(reports, report)
	}
	return map[string]any{"completed": complete, "partial": partial, "reports": reports}
}

func newFidelityCache(t *testing.T) fidelityCacheDriver {
	resetCompletionRecordsCache()
	t.Cleanup(resetCompletionRecordsCache)
	advance := withFrozenTime(t, time.Date(2026, 11, 1, 0, 0, 0, 0, time.UTC))
	status := 0
	lister := &fakeLister{respond: func(token string) (*completionRecordPage, error) {
		if status != 0 {
			return nil, &appPlatformUpstreamError{status: status, msg: "injected"}
		}
		return &completionRecordPage{}, nil
	}}
	return fidelityCacheDriver{
		advance:    advance,
		invalidate: func() { invalidateCompletionIndex("stacks-1") },
		read: func(forced bool, nextStatus int) fidelityCacheResult {
			status = nextStatus
			index, err := getCompletionIndex(context.Background(), "stacks-1", lister, forced, log.DefaultLogger)
			result := fidelityCacheResult{Calls: lister.callCount(), Failed: err != nil}
			if index != nil {
				result.AsOf = index.asOf.Format(time.RFC3339Nano)
			}
			return result
		},
	}
}

func fidelityCompletionRead(t *testing.T, status int, capability bool) *httptest.ResponseRecorder {
	lister := &fakeLister{respond: func(token string) (*completionRecordPage, error) {
		if status == -3 {
			raw, next := fidelityReadPage(token)
			page := &completionRecordPage{Continue: next}
			for _, value := range raw {
				var record completionRecordSpec
				if err := json.Unmarshal(value, &record); err != nil {
					t.Fatal(err)
				}
				record.LastUpdatedAt = recordLastUpdatedAt(nil, record.RecordedAt)
				page.Records = append(page.Records, record)
			}
			return page, nil
		}
		if status == -1 {
			var value any
			return nil, json.Unmarshal([]byte("not-json"), &value)
		}
		if status == -4 {
			return nil, fidelityFailure("cancelled")
		}
		if status == -5 {
			return nil, fidelityFailure("network")
		}
		if status == -2 {
			return nil, context.DeadlineExceeded
		}
		if status != 200 {
			return nil, &appPlatformUpstreamError{status: status, msg: "injected"}
		}
		return &completionRecordPage{}, nil
	}}
	withLister(t, lister)
	request := completionRequest(t, "/completion-records/my", "user:abc")
	response := httptest.NewRecorder()
	app := newTestApp(t)
	if capability {
		app.handleCompletionCapability(response, request)
	} else {
		app.handleMyCompletions(response, request)
	}
	return response
}

func fidelitySettingsRead(t *testing.T, status int, body string) *httptest.ResponseRecorder {
	exchange := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		_, _ = w.Write([]byte(`{"status":"success","data":{"token":"obo-token"}}`))
	}))
	defer exchange.Close()
	exchanger, err := auth.New("cap-token", exchange.URL)
	if err != nil {
		t.Fatal(err)
	}
	upstream := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path == auth.SigningKeysPath {
			_, _ = w.Write(jwksBody(testSigningKeyID, testSigningKey()))
			return
		}
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(status)
		_, _ = w.Write([]byte(body))
	}))
	defer upstream.Close()
	app := &App{logger: log.DefaultLogger, oboExchanger: exchanger}
	response := httptest.NewRecorder()
	app.handlePathfinderSettings(response, customGuideRequestWithConfig(t, "/pathfinder-settings", "user:1", map[string]string{sdkconfig.AppURL: upstream.URL}))
	return response
}
